cmake_minimum_required(VERSION 3.24)
project(VocalStudio VERSION 1.4.1 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VOCAL_BUILD_TESTS "Build deterministic DSP and real host integration tests" ON)
include(FetchContent)
FetchContent_Declare(JUCE URL https://github.com/juce-framework/JUCE/archive/51a8a6d7aeae7326956d747737ccf1575e61e209.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(JUCE)
if(WIN32)
  FetchContent_Declare(asiosdk URL https://github.com/langlauf/asiosdk/archive/c1d68b74d8298f50cffab44f915a9977d1952576.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(asiosdk)
endif()
add_library(VocalDsp STATIC Source/Dsp.cpp Source/Pitch.cpp Source/VoiceFit.cpp Source/LiveMusic.cpp Source/AutoEq.cpp)
target_include_directories(VocalDsp PUBLIC Source PRIVATE ThirdParty)
if(WIN32)
  # Current CEF's sandbox bootstrap loads the app as a DLL before starting any
  # untrusted subprocess. Keep JUCE's main app in the same sandboxed lifecycle.
  add_library(VocalStudio SHARED)
  target_compile_definitions(VocalStudio PRIVATE JUCE_STANDALONE_APPLICATION=1 JUCE_APPLICATION_NAME_STRING="VocalStudio" JUCE_APPLICATION_VERSION_STRING="1.4.1")
  set_target_properties(VocalStudio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VocalStudio_artefacts/$<CONFIG>")
else()
  juce_add_gui_app(VocalStudio VERSION 1.4.1 PRODUCT_NAME "VocalStudio" COMPANY_NAME "VocalStudio" BUNDLE_ID "studio.vocal.personal")
endif()
target_sources(VocalStudio PRIVATE Source/Main.cpp Source/YouTubeBrowser.cpp Source/KaraokePage.cpp Source/Engine.cpp Source/EnginePitch.cpp Source/EngineTone.cpp Source/PluginTone.cpp Source/MainComponent.cpp Source/MainAutoEq.cpp Source/PolishPanel.cpp Source/AutoEqPanel.cpp Source/MainFit.cpp Source/VocalTools.cpp Source/TuneWorkflow.cpp Tests/EngineTests.cpp)
target_compile_definitions(VocalStudio PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DIRECTSOUND=0)
if(WIN32)
  target_compile_definitions(VocalStudio PRIVATE JUCE_ASIO=1)
  target_include_directories(VocalStudio PRIVATE "${asiosdk_SOURCE_DIR}/common")
endif()
target_link_libraries(VocalStudio PRIVATE VocalDsp juce::juce_audio_utils juce::juce_audio_processors juce::juce_gui_extra PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags)
if(WIN32)
  include(cmake/Browser.cmake)
endif()
if(MSVC)
  target_compile_options(VocalStudio PRIVATE /utf-8)
  target_compile_options(VocalDsp PRIVATE /utf-8)
endif()
if(VOCAL_BUILD_TESTS)
  enable_testing()
  add_executable(VocalPolishTests Tests/PolishTests.cpp)
  target_link_libraries(VocalPolishTests PRIVATE VocalDsp)
  add_test(NAME VocalPolishTests COMMAND VocalPolishTests)
  add_executable(VocalAutoEqTests Tests/AutoEqTests.cpp)
  target_link_libraries(VocalAutoEqTests PRIVATE VocalDsp)
  add_test(NAME VocalAutoEqTests COMMAND VocalAutoEqTests)
  add_executable(VocalDspTests Tests/DspTests.cpp)
  target_link_libraries(VocalDspTests PRIVATE VocalDsp)
  add_test(NAME VocalDspTests COMMAND VocalDspTests)
  add_executable(VocalPitchTests Tests/PitchTests.cpp)
  target_link_libraries(VocalPitchTests PRIVATE VocalDsp)
  add_test(NAME VocalPitchTests COMMAND VocalPitchTests)
  add_executable(VocalLiveMusicTests Tests/LiveMusicTests.cpp)
  target_link_libraries(VocalLiveMusicTests PRIVATE VocalDsp)
  add_test(NAME VocalLiveMusicTests COMMAND VocalLiveMusicTests)
  add_executable(VocalVoiceFitTests Tests/VoiceFitTests.cpp)
  target_link_libraries(VocalVoiceFitTests PRIVATE VocalDsp)
  add_test(NAME VocalVoiceFitTests COMMAND VocalVoiceFitTests)
  juce_add_plugin(VocalStudioTestPlugin COMPANY_NAME "VocalStudio" PLUGIN_MANUFACTURER_CODE Vstu PLUGIN_CODE Vstf FORMATS VST3 PRODUCT_NAME "VocalStudio Test Gain" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(VocalStudioTestPlugin PRIVATE Tests/FixturePlugin.cpp)
  target_compile_definitions(VocalStudioTestPlugin PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(VocalStudioTestPlugin PRIVATE juce::juce_audio_processors juce::juce_gui_extra PUBLIC juce::juce_recommended_config_flags)
  add_test(NAME VocalEngineTests COMMAND "$<TARGET_FILE_DIR:VocalStudio>/VocalStudio.exe" --self-test --fixture "$<TARGET_FILE:VocalStudioTestPlugin_VST3>" --report "${CMAKE_BINARY_DIR}/EngineSelfTest.json")
  set_tests_properties(VocalDspTests VocalPitchTests VocalVoiceFitTests VocalEngineTests PROPERTIES TIMEOUT 120)
endif()
